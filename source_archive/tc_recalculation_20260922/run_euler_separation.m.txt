function run_euler_separation()
% Independent TC trials, fixed before observing coverage; no target shifts.
code=fileparts(mfilename('fullpath'));addpath(code);
root=fullfile(code,'results','euler_target_pool_separation_20260923');
if ~isfolder(root),mkdir(root);end
maxNumCompThreads(2);setenv('OPENBLAS_NUM_THREADS','2');setenv('OMP_NUM_THREADS','2');
old=fullfile(code,'results','paper_scale_original_grid_20260922');
s=load(fullfile(old,'linear_rqmc_n768','design.mat'),'design');d=s.design;
s=load(fullfile(old,'reference_linear','reference_coverage_grid.mat'),'ref');ref=s.ref;
assert(d.n==768&&size(d.base,2)==5&&abs(sum(d.P)-1)<1e-12&&ref.h==.08);
grid=ref.y;h=ref.h;M=5000;Rlist=[32,128];B=399;seed=2026092302;
config=struct('M',M,'R',Rlist,'B',B,'seed',seed,'n',768,'h',h,'grid',grid, ...
    'centering',false,'fresh_curves',true,'nested_R_prefix',true);
lock=fullfile(root,'config.mat');
if isfile(lock),x=load(lock);assert(isequaln(x.config,config));else,save(lock,'config','-v7');end
j=1:5;a=sin(j*pi/2)./(j*pi).^2;a=a/norm(a);
for first=1:50:M
    last=min(first+49,M);file=fullfile(root,sprintf('batch_%05d_%05d.mat',first,last));
    if isfile(file),continue;end
    timer=tic;L=last-first+1;G=numel(grid);
    lower=zeros(L,2,3,G);upper=lower;qSaved=zeros(128,G,L);
    zero=zeros(L,2,G);ties=zero;
    for trial=first:last
        stream=RandStream('Threefry',Seed=seed);stream.Substream=trial;
        shifts=rand(stream,128,5);response=zeros(768,128);
        for dim=1:5
            u=mod(d.base(:,dim)+shifts(:,dim).',1);
            u(u==0)=realmin;u(u==1)=1-eps/2;
            response=response+a(dim)*norminv(u);
        end
        H=exp(-.5*((grid-response(:))/h).^2)/(sqrt(2*pi)*h);
        q=reshape(sum(reshape(H,768,128,G).*reshape(d.P,768,1,1),1),128,G);
        if trial==1
            % Compare vectorized response and quadrature to the original loop.
            for k=1:3
                u=mod(d.base+shifts(k,:),1);u(u==0)=realmin;u(u==1)=1-eps/2;
                y=norminv(u)*a.';
                expected=d.P.'*(exp(-.5*((grid-y)/h).^2)/(sqrt(2*pi)*h));
                assert(max(abs(q(k,:)-expected))<1e-12);
            end
        end
        idx=trial-first+1;qSaved(:,:,idx)=q;
        for ir=1:2
            out=tc_intervals_from_original(q(1:Rlist(ir),:),B,.05, ...
                1000000+trial*2+ir,seed);
            lower(idx,ir,:,:)=out.lower;upper(idx,ir,:,:)=out.upper;
            zero(idx,ir,:)=out.zero_bootstrap_variance;ties(idx,ir,:)=out.bootstrap_mean_ties;
        end
    end
    seconds=toc(timer);tmp=[char(file),'.writing.mat'];
    save(tmp,'lower','upper','qSaved','first','last','seconds','zero','ties','-v7');
    movefile(tmp,file);
    fprintf('Euler independent trials %d/%d; batch seconds %.2f\n',last,M,seconds);
end
fprintf('EULER INDEPENDENT TRIALS COMPLETE: %s\n',root);
end
